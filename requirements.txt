# CZN Launcher Lite —— 运行依赖
# ---------------------------------------------------------------------------
# 平台：Windows 10/11 x64
#   本项目使用 winreg / 命名管道 / ShellExecuteExW，无法在其他平台运行。
#
# 用法：
#   pip install -r requirements.txt
#   python gui_flet.py    # 图形界面（Flet / Material 3，主推）
#   python gui.py         # 图形界面（customtkinter 旧版，保留兜底）
#   python czn_lite.py    # 控制台版（同等功能）
#
# 注意：打包工具 PyInstaller 不在此列 —— build*.bat 打包时会自行安装，
#       只想跑源码的人不需要它。
# ---------------------------------------------------------------------------

# HTTP 客户端。本项目依赖其 TLS 指纹模拟（impersonate）行为，
# 该行为在不同小版本间会变化，故锁在实测的 0.16.x
curl_cffi>=0.16.3,<0.17

# RSA 公钥解密 + AES-CBC 加解密（管道握手 2000 / 2001 帧）
pycryptodome>=3.17,<4.0

# Win32 API 绑定：命名管道、注册表、ShellExecuteExW
pywin32>=306; sys_platform == "win32"

# GUI 组件库（Flet 版）+ 桌面客户端。三个钉在 1.0.3：build_flet.bat 打进包里的
# 就是 flet v1.0.3 的 flet-windows.zip，Python 侧与 Flutter 视图必须同一个
# release；换 vendor 里的 zip 才跟着动这里。
# flet-desktop 不能省：缺件时打包只会 WARNING，随包客户端照样落地，
# 死的是启动时的冻结件。
flet==1.0.3
flet-video==1.0.3
flet-desktop==1.0.3

# 二维码 png 显示与验证码图合成。下界 13.0：12.3.0 及更早的 28 条解码器
# CVE 只在 13.0.0 修，而这两处吃的都是服务端下发的字节。
pillow>=13.0,<14

# curl_cffi 的 CA 信任库取自 certifi.where()。下界 2026.10.19：更早的版本
# 带着 eMudhra 已吊销私钥的 emSign ECC Root CA - G3，而本程序对 onstove
# 域名没有证书固定。
certifi>=2026.10.19
